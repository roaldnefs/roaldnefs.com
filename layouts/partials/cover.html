{{- /* Generates a cover image as an SVG file and returns its URL. Call with a string, usually a page title. The picture is derived from that string, so every title gets its own and it stays the same between builds. Scenes: 0 = streaks along a horizon, 1 = beams rising from the horizon, 2 = a burst of lines from one point, 3 = rings spreading from one point. */ -}} {{- $seed := . -}} {{- $n := hash.FNV32a $seed "#8fc8ff" "#3f8fe0" "#d9ebff" -}} {{- $hx := add 380 (mod (div $n 16) 840) -}} {{- $hy := add 470 (mod (div $n 128) 150) -}} {{- if ge $scene 2 }}{{ $hy = add 330 (mod (div $n 128) 240) }}{{ end -}} {{- $marks := slice -}} {{- if eq $scene 3 -}} {{- range $i := seq 1 9 -}} {{- $r := mod (hash.FNV32a (print $seed "r" $i)) 1000 -}} {{- $rx := add (mul $i 62) (div $r 14) -}} {{- $color := index $warm (mod $r 5) -}} {{- if eq (mod $i 3) 0 }}{{ $color = index $cool (mod $r 3) }}{{ end -}} {{- $marks = $marks | append (printf `` $hx $hy $rx (div (mul $rx 38) 100) $color (add 2.5 (div (float (mod $r 60)) 10)) (sub 0.9 (mul (float $i) 0.08))) -}} {{- end -}} {{- else -}} {{- range $i := seq 1 46 -}} {{- $a := mod (hash.FNV32a (print $seed "a" $i)) 1000 -}} {{- $b := mod (hash.FNV32a (print $seed "b" $i)) 1000 -}} {{- $c := mod (hash.FNV32a (print $seed "c" $i)) 1000 -}} {{- $color := index $warm (mod $a 5) -}} {{- if eq (mod $i 6) 0 }}{{ $color = index $cool (mod $a 3) }}{{ end -}} {{- 14) $hy $w $h $color $opacity $angle $hx $hy) -}} {{- else -}} {{- /* Lines hug the horizon; most are short, a few run long */ -}} {{- $w := add 90 (div (mul $a $a) 1300) -}} {{- $spread := sub (div end -}} {{- end -}} {{- $turn := "" -}} {{- if eq $scene 1 }}{{ $turn = printf ` transform="rotate(-90 %d %d)"` $hx $hy }}{{ end -}} {{- $smear := "9 0.5" -}} {{- if eq $scene 2 }}{{ $smear = "2" }}{{ end -}} {{- if eq $scene 3 }}{{ $smear = "3" }}{{ end -}} {{- $svg := printf `` (index $p 0) (sub (div (float $hy) 900) 0.04) (index $p 1) (add (div (float $hy) 900) 0.03) (index $p 2) (index $p 3) $file.RelPermalink -}}
